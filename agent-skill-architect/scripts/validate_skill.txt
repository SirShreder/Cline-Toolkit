Bash structural validator for AgentSkills.io skill packages.
Save as an executable script (e.g. scripts/validate_skill.sh) or run directly:
    bash scripts/validate_skill.sh /path/to/<skill-directory>

The script is embedded below the cut-here marker because the authoring registry
accepts only SKILL.md and Markdown/plain-text/CSV/JSON/YAML support files; in a
deployment that permits executable files, save the extracted section as an
executable script.

Checks:
  1. The skill directory and SKILL.md exist (SKILL.md non-empty).
  2. YAML frontmatter opens with '---' and is closed by a '---' line.
  3. Frontmatter keys are restricted to name, description, license,
     allowed-tools, metadata, and compatibility; any other key fails
     (user-created registry skills must not add fields such as visibility
     or defaultEnabled).
  4. 'name' is non-empty, lowercase letters/digits/hyphens only, max 64 chars,
     and matches the skill directory name (Cline requires an exact match).
  5. 'description' is non-empty, at most 1024 characters, and contains no
     angle brackets (< or >; some runtimes render the catalog as markup).
     Above 500 characters a warning is printed: 500 is the recommended budget
     for all runtimes; the DeepSeek harness session catalog hard-caps
     descriptions at 500 characters by default.
  6. The optional 'compatibility' field, when present, is a single string of
     at most 500 characters (lists and mappings fail).
  7. Every relative path under references/, scripts/, assets/, docs/, or
     templates/ that is referenced in any of the skill's Markdown files
     resolves to an existing file or directory (Ghost File Rule). docs/ and
     templates/ are Cline's documented layouts; URLs are ignored when matching.
  8. Warning when SKILL.md exceeds 500 lines (cross-runtime context budget
     guidance: under 5k tokens, roughly 500 lines).

Frontmatter values may be plain scalars (name: my-skill) or YAML block
scalars (name: | followed by indented lines). Multi-line values are joined
with single spaces.

Exit codes: 0 = valid (warnings possible), 1 = validation errors found, 2 = usage error.

Authoring-decision rules in SKILL.md (compose edits against on-disk state, remediate
by failure class, layer ownership of enforcement mechanisms, three-way code placement
— code read to reason in references/, verbatim copy targets in assets/ or templates/,
executable code in scripts/, always named as concrete relative paths — fail-closed
validators, circular reference chains, priority-chain contradiction resolution,
stale-content sweeps, one task per section / one instruction per sentence) are
procedural judgments and are not structurally checkable; they are intentionally not
implemented in this script. The description-to-body fidelity check
(references/AUDIT.md, item 19) is likewise semantic — the validator checks only the
description's schema, length, and bracket constraints.

#!/usr/bin/env bash
set -u

if [ "$#" -ne 1 ]; then
  echo "Usage: bash validate_skill.sh /path/to/<skill-directory>" >&2
  exit 2
fi

SKILL_DIR="$1"
errors=0
warnings=0

fail() { printf 'ERROR: %s\\n' "$1" >&2; errors=$((errors + 1)); }
warn() { printf 'WARNING: %s\\n' "$1" >&2; warnings=$((warnings + 1)); }
pass() { printf 'OK: %s\\n' "$1"; }

# get_field <field-name>  — reads frontmatter from stdin, prints the value.
# Supports plain scalars and block scalars (|, >, |-, >-); multi-line
# block values are joined with single spaces.
get_field() {
  awk -v f="$1" '
    inblock {
      if ($0 ~ /^[[:space:]]/) {
        line = $0; sub(/^[[:space:]]+/, "", line)
        buf = buf sep line; sep = " "; next
      }
      print buf; inblock = 0; exit
    }
    index($0, f ":") == 1 {
      val = substr($0, length(f) + 2)
      sub(/^[[:space:]]+/, "", val)
      if (val == "|" || val == ">" || val == "|-" || val == ">-") {
        inblock = 1; buf = ""; sep = ""; next
      }
      print val; exit
    }
    END { if (inblock) print buf }
  '
}

if [ ! -d "$SKILL_DIR" ]; then
  fail "Skill directory not found: $SKILL_DIR"
  exit 1
fi

SKILL_MD="$SKILL_DIR/SKILL.md"

# --- Check 1: SKILL.md exists and is non-empty -------------------------------
if [ ! -f "$SKILL_MD" ] || [ ! -s "$SKILL_MD" ]; then
  fail "SKILL.md is missing or empty at: $SKILL_MD"
  printf 'FAILED: %s validation error(s).\\n' "$errors" >&2
  exit 1
fi
pass "SKILL.md exists and is non-empty"

# --- Check 2: frontmatter block ----------------------------------------------
first_line=$(head -n 1 "$SKILL_MD")
if [ "$first_line" = "---" ]; then
  pass "Frontmatter opens with '---'"
else
  fail "SKILL.md must start with a '---' line (found: '$first_line')"
fi

if awk 'NR > 1 && /^---[[:space:]]*$/ { found = 1; exit } END { exit (found ? 0 : 1) }' "$SKILL_MD"; then
  pass "Frontmatter is closed by a '---' line"
else
  fail "Frontmatter is never closed by a '---' line"
fi

frontmatter=$(awk 'NR == 1 { next } /^---[[:space:]]*$/ { exit } { print }' "$SKILL_MD")

# --- Check 2b: frontmatter keys are whitelisted --------------------------------
bad_keys=$(printf '%s\\n' "$frontmatter" | awk -F':' '
  /^[A-Za-z0-9_-]+:/ {
    k = $1
    if (k != "name" && k != "description" && k != "license" && k != "allowed-tools" && k != "metadata" && k != "compatibility")
      print k
  }')
if [ -n "$bad_keys" ]; then
  fail "Unexpected frontmatter key(s): $(printf '%s ' $bad_keys)allowed: name, description, license, allowed-tools, metadata, compatibility"
else
  pass "frontmatter keys are whitelisted"
fi

# --- Check 3: name ------------------------------------------------------------
name=$(printf '%s\\n' "$frontmatter" | get_field name)
name=${name#\"}; name=${name%\"}
if [ -z "$name" ]; then
  fail "Frontmatter field 'name' is missing or empty"
elif [ "${#name}" -gt 64 ]; then
  fail "Frontmatter 'name' exceeds 64 characters: $name"
elif printf '%s' "$name" | grep -Eq '^[a-z0-9]+(-[a-z0-9]+)*$'; then
  pass "name is valid: $name"
else
  fail "Frontmatter 'name' must be lowercase letters, digits, and single hyphens: $name"
fi

# --- Check 3b: name matches the directory name --------------------------------
dir_name=$(basename "$SKILL_DIR")
if [ -n "$name" ]; then
  if [ "$name" = "$dir_name" ]; then
    pass "name matches the directory name: $dir_name"
  else
    fail "Frontmatter 'name' ($name) must match the skill directory name: $dir_name"
  fi
fi

# --- Check 4: description -----------------------------------------------------
desc=$(printf '%s\\n' "$frontmatter" | get_field description)
desc=${desc#\"}; desc=${desc%\"}
if [ -z "$desc" ]; then
  fail "Frontmatter field 'description' is missing or empty"
else
  if [ "${#desc}" -gt 1024 ]; then
    fail "Frontmatter 'description' exceeds 1024 characters (${#desc})"
  elif [ "${#desc}" -gt 500 ]; then
    warn "description is ${#desc} characters; recommended budget is 500 for all runtimes (the DeepSeek harness session catalog hard-caps at 500 by default)"
  fi
  case "$desc" in
    *'<'*|*'>'*) fail "Frontmatter 'description' must not contain angle brackets (< or >): some runtimes render the catalog as markup" ;;
  esac
  pass "description is present (${#desc} characters)"
fi

# --- Check 4b: compatibility (optional) ----------------------------------------
compat=$(printf '%s\\n' "$frontmatter" | get_field compatibility)
if [ -n "$compat" ]; then
  case "$compat" in
    '['*|'{'*) fail "Frontmatter 'compatibility' must be a single string, not a list or mapping" ;;
    *)
      if [ "${#compat}" -gt 500 ]; then
        fail "Frontmatter 'compatibility' exceeds 500 characters (${#compat})"
      else
        pass "compatibility is present (${#compat} characters)"
      fi
      ;;
  esac
fi

# --- Check 5: referenced paths resolve (Ghost File Rule) ----------------------
# URLs are stripped first so links to documentation (e.g. GitHub paths
# containing docs/) are not mistaken for bundled-file references.
while IFS= read -r -d '' md_file; do
  while IFS= read -r ref; do
    if [ -e "$SKILL_DIR/$ref" ]; then
      pass "path resolves: $ref"
    else
      fail "Referenced path does not exist: $ref (in $md_file)"
    fi
  done < <(sed -E 's#https?://[^)[:space:]]+##g' "$md_file" 2>/dev/null | grep -Eo '(references|scripts|assets|docs|templates)/[A-Za-z0-9._/-]+' | sed -E 's/[.,;:)]+$//' | sort -u)
done < <(find "$SKILL_DIR" -type f -name '*.md' -print0 2>/dev/null)

# --- Check 6: SKILL.md context budget (warning only) --------------------------
skill_lines=$(wc -l < "$SKILL_MD")
if [ "$skill_lines" -gt 500 ]; then
  warn "SKILL.md is $skill_lines lines; keep it under 500 lines (~5k tokens) — move detail into reference files"
fi

# --- Summary -------------------------------------------------------------------
if [ "$errors" -gt 0 ]; then
  printf 'FAILED: %s validation error(s).\\n' "$errors" >&2
  exit 1
fi
if [ "$warnings" -gt 0 ]; then
  printf 'VALID with %s warning(s): skill package passes structural checks.\\n' "$warnings"
else
  printf 'VALID: skill package passes structural checks.\\n'
fi
exit 0
